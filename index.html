import pygame pygame.init() screen = pygame.display.set_mode((900, 600)) pygame.display.set_caption("PORTAL MAYHEM") clock = pygame.time.Clock() world = "Обычная вселенная" player = pygame.Rect(100, 250, 40, 60) portal = None worlds = ["Рик и Морти", "Minecraft", "Roblox", "Космос"] world_index = 0 font = pygame.font.SysFont("Arial", 28) running = True while running: screen.fill((25, 25, 35)) for event in pygame.event.get(): if event.type == pygame.QUIT: running = False if event.type == pygame.KEYDOWN: if event.key == pygame.K_w: # Создаём портал, но не меняем мир world_index = (world_index + 1) % len(worlds) portal = pygame.Rect(700, 220, 50, 150) keys = pygame.key.get_pressed() if keys[pygame.K_LEFT]: player.x -= 5 if keys[pygame.K_RIGHT]: player.x += 5 if keys[pygame.K_UP]: player.y -= 5 if keys[pygame.K_DOWN]: player.y += 5 # Проверяем пересечение с порталом if portal and player.colliderect(portal): world = worlds[world_index] portal = None player.x = 100 # Игрок pygame.draw.rect(screen, (255, 255, 255), player) # Портал if portal: pygame.draw.ellipse(screen, (0, 255, 100), portal) # Информация screen.blit(font.render("WORLD: " + world, True, (255, 255, 255)), (20, 20)) screen.blit(font.render("Нажми W для создания портала", True, (255, 255, 255)), (20, 60)) pygame.display.flip() clock.tick(60) pygame.quit()
